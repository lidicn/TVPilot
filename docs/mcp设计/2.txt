def _screenshot_to_url_result(data: Any) -> Dict[str, Any]:
    width = 426
    height = 240
    if isinstance(data, Mapping):
        width = int(data.get("width") or data.get("w") or width)
        height = int(data.get("height") or data.get("h") or height)
        for k in ("url", "image_url", "screenshot_url"):
            if data.get(k):
                return {"url": str(data[k]), "width": width, "height": height}
        b64 = None
        for k in ("base64", "b64", "image", "data", "jpeg", "png"):
            v = data.get(k)
            if isinstance(v, str) and len(v) > 100:
                b64 = v
                break
        path = data.get("path") or data.get("file")
        if path and os.path.exists(str(path)):
            return _save_file_url(str(path), width, height)
    elif isinstance(data, str):
        if data.startswith("http://") or data.startswith("https://") or data.startswith("file://"):
            return {"url": data, "width": width, "height": height}
        if os.path.exists(data):
            return _save_file_url(data, width, height)
        b64 = data
    if b64:
        return _save_base64(b64, width, height)
    raise ToolError("ERR_SCREENSHOT_FORMAT", "截图结果既不是 URL 也不是可保存的 base64/路径")

def _save_base64(b64: str, width: int, height: int) -> Dict[str, Any]:
    if "," in b64 and b64.strip().startswith("data:"):
        b64 = b64.split(",", 1)[1]
    try:
        raw = base64.b64decode(b64, validate=False)
    except Exception as exc:
        raise ToolError("ERR_SCREENSHOT_BASE64", f"base64 解码失败: {exc}")
    fn = f"shot_{int(time.time()*1000)}.jpg"
    path = os.path.join(_SHOT_DIR, fn)
    with open(path, "wb") as f:
        f.write(raw)
    return _save_file_url(path, width, height)

def _save_file_url(path: str, width: int, height: int) -> Dict[str, Any]:
    fn = os.path.basename(path)
    dst = os.path.join(_SHOT_DIR, fn)
    if os.path.abspath(path) != os.path.abspath(dst):
        with open(path, "rb") as src, open(dst, "wb") as out:
            out.write(src.read())
    port = _current_port()
    if port:
        url = f"http://{DEFAULT_HOST}:{port}{_SHOT_URL_PREFIX}{fn}"
    else:
        url = f"file://{dst}"

class McpServer:
    def __init__(self, host=DEFAULT_HOST, port=DEFAULT_PORT):
        self.host = host
        self.port = port
        self._httpd = None

    def handle_jsonrpc(self, req: Any) -> Optional[Dict[str, Any]]:
        if isinstance(req, list):
            # 批量
            responses = [self._handle_one(r) for r in req]
            responses = [r for r in responses if r is not None]
            return responses if responses else None
        return self._handle_one(req)

    def _handle_one(self, req):
        if not isinstance(req, Mapping):
            return _error(None, -32600, "Invalid Request")
        rid = req.get("id")
        method = req.get("method")
        params = req.get("params") or {}
        if req.get("jsonrpc") != "2.0" or not isinstance(method, str):
            return _error(rid, -32600, "Invalid Request")
        if method == "initialize":
            return _ok(rid, {
                "protocolVersion": PROTOCOL_VERSION,
                "capabilities": {"tools": True},
                "serverInfo": {"name": SERVER_NAME, "version": SERVER_VERSION},
            })
        if method == "notifications/initialized":
            return None
        if method == "ping":
            return _ok(rid, {})
        if method == "tools/list":
            return _ok(rid, {"tools": TOOLS})
        if method == "tools/call":
            return self._handle_tools_call(rid, params)
        return _error(rid, -32601, f"Method not found: {method}")

    def _handle_tools_call(self, rid, params):
        if not isinstance(params, Mapping):
            return _error(rid, -32602, "Invalid params")
        name = params.get("name")
        arguments = params.get("arguments") or {}
        if not isinstance(arguments, Mapping):
            return _error(rid, -32602, "arguments 必须是对象")
        handler = _TOOL_HANDLERS.get(name)
        if handler is None:
            return _error(rid, -32602, f"Unknown tool: {name}")
        try:
            data = handler(dict(arguments))
            text = _json_dumps(data)
            result = {
                "content": [{"type": "text", "text": text}],
                "structuredContent": data,
                "isError": False,
            }
            if isinstance(data, Mapping):
                # 兼容验收：顶层也暴露关键字段
                for k, v in data.items():
                    result.setdefault(k, v)
            return _ok(rid, result)
        except ToolError as exc:
            return _ok(rid, {
                "content": [{"type": "text", "text": exc.message}],
                "isError": True,
                "errorCode": exc.code,
                "error": exc.message,
            })
        except Exception as exc:
            return _ok(rid, {
                "content": [{"type": "text", "text": f"{type(exc).__name__}: {exc}"}],
                "isError": True,
                "error": str(exc),
            })

    err = {"code": code, "message": message}

class _McpHTTPHandler(BaseHTTPRequestHandler):
    server_version = "TVPilotMCP/0.1"
    protocol_version = "HTTP/1.1"

    def log_message(self, fmt, *args):
        sys.stderr.write("[tvpilot-mcp] " + fmt % args + "\n")

    def _read_body(self):
        length = int(self.headers.get("Content-Length") or 0)
        return self.rfile.read(length) if length else b""

    def _send_json(self, obj, status=200):
        body = _json_dumps(obj).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _send_sse(self, obj, status=200):
        data = _json_dumps(obj)
        body = f"event: message\ndata: {data}\n\n".encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "text/event-stream; charset=utf-8")
        self.send_header("Cache-Control", "no-cache")
        self.send_header("Connection", "keep-alive")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        parsed = urlparse(self.path)
        if parsed.path.startswith(_SHOT_URL_PREFIX):
            fn = os.path.basename(parsed.path[len(_SHOT_URL_PREFIX):])
            path = os.path.join(_SHOT_DIR, fn)
            if not os.path.exists(path):
                self.send_error(404)
                return
            with open(path, "rb") as f:
                body = f.read()
            self.send_response(200)
            ctype = "image/jpeg"
            if fn.lower().endswith(".png"):
                ctype = "image/png"
            self.send_header("Content-Type", ctype)
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
            return
        if parsed.path == "/mcp":
            # 简单 SSE 保活，或返回能力说明
            self._send_sse({"jsonrpc":"2.0","method":"notifications/initialized","params":{}})
            return
        self.send_error(404)

    def do_POST(self):
        parsed = urlparse(self.path)
        if parsed.path != "/mcp":
            self.send_error(404)
            return
        try:
            raw = self._read_body()
            req = json.loads(raw.decode("utf-8"))
        except Exception:
            self._send_json(_error(None, -32700, "Parse error"), 400)
            return
        server = self.server.mcp_server  # type: ignore
        resp = server.handle_jsonrpc(req)
        if resp is None:
            self.send_response(202)
            self.send_header("Content-Length", "0")
            self.end_headers()
            return
        accept = self.headers.get("Accept", "")
        if "text/event-stream" in accept:
            self._send_sse(resp)
        else:
            self._send_json(resp)

def serve_stdio():
    server = McpServer()

def _save_base64(b64: str, width: int, height: int) -> Dict[str, Any]:
    if b64.strip().startswith("data:") and "," in b64:
        b64 = b64.split(",", 1)[1]
    try:
        raw = base64.b64decode(b64, validate=False)
    except Exception as exc:  # noqa: BLE001
        raise ToolError("ERR_SCREENSHOT_BASE64", f"base64 解码失败: {exc}") from exc

    fn = f"shot_{int(time.time() * 1000)}.jpg"
    path = os.path.join(_SHOT_DIR, fn)
    with open(path, "wb") as f:
        f.write(raw)
    return _save_file_url(path, width, height)


def _save_file_url(path: str, width: int, height: int) -> Dict[str, Any]:
    fn = os.path.basename(path)
    dst = os.path.join(_SHOT_DIR, fn)
    if os.path.abspath(path) != os.path.abspath(dst):
        with open(path, "rb") as src, open(dst, "wb") as out:
            out.write(src.read())

    port = _current_port()
    if port:
        url = f"http://{DEFAULT_HOST}:{port}{_SHOT_URL_PREFIX}{fn}"
    else:
        url = f"file://{dst}"
    return {"url": url, "width": int(width), "height": int(height)}


def _screenshot_to_url_result(data: Any) -> Dict[str, Any]:
    """把 tool_screenshot 的返回转成 URL 结果，绝不回传 base64。"""
    width = 426
    height = 240

    if isinstance(data, Mapping):
        try:
            width = int(data.get("width") or data.get("w") or width)
        except Exception:
            pass
        try:
            height = int(data.get("height") or data.get("h") or height)
        except Exception:
            pass

        for key in ("url", "image_url", "screenshot_url"):
            if data.get(key):
                return {"url": str(data[key]), "width": width, "height": height}

        path = data.get("path") or data.get("file")
        if path and os.path.exists(str(path)):
            return _save_file_url(str(path), width, height)

        for key in ("base64", "b64", "image", "data", "jpeg", "png"):
            val = data.get(key)
            if isinstance(val, str) and len(val) > 32:
                return _save_base64(val, width, height)

    if isinstance(data, str):
        if data.startswith(("http://", "https://", "file://")):
            return {"url": data, "width": width, "height": height}
        if os.path.exists(data):
            return _save_file_url(data, width, height)
        return _save_base64(data, width, height)

    raise ToolError("ERR_SCREENSHOT_FORMAT", "截图结果既不是 URL，也不是可保存的 base64/路径")

def _ok(rid: Any, result: Any) -> Dict[str, Any]:
    return {"jsonrpc": "2.0", "id": rid, "result": result}


def _error(rid: Any, code: int, message: str, data: Any = None) -> Dict[str, Any]:
    err: Dict[str, Any] = {"code": code, "message": message}
    if data is not None:
        err["data"] = data
    return {"jsonrpc": "2.0", "id": rid, "error": err}


class McpServer:
    def __init__(self, host: str = DEFAULT_HOST, port: int = DEFAULT_PORT):
        self.host = host
        self.port = port

    def handle_jsonrpc(self, req: Any) -> Optional[Any]:
        if isinstance(req, list):
            responses = [self._handle_one(x) for x in req]
            responses = [r for r in responses if r is not None]
            return responses if responses else None
        return self._handle_one(req)

    def _handle_one(self, req: Any) -> Optional[Dict[str, Any]]:
        if not isinstance(req, Mapping):
            return _error(None, -32600, "Invalid Request")

        rid = req.get("id")
        method = req.get("method")
        params = req.get("params") or {}

        if req.get("jsonrpc") != "2.0" or not isinstance(method, str):
            return _error(rid, -32600, "Invalid Request")

        if method == "initialize":
            return _ok(
                rid,
                {
                    "protocolVersion": PROTOCOL_VERSION,
                    "capabilities": {"tools": True},
                    "serverInfo": {"name": SERVER_NAME, "version": SERVER_VERSION},
                },
            )

        if method == "notifications/initialized":
            return None

        if method == "ping":
            return _ok(rid, {})

        if method == "tools/list":
            return _ok(rid, {"tools": TOOLS})

        if method == "tools/call":
            return self._handle_tools_call(rid, params)

        return _error(rid, -32601, f"Method not found: {method}")

    def _handle_tools_call(self, rid: Any, params: Any) -> Dict[str, Any]:
        if not isinstance(params, Mapping):
            return _error(rid, -32602, "Invalid params")

        name = params.get("name")
        arguments = params.get("arguments") or {}
        if not isinstance(arguments, Mapping):
            return _error(rid, -32602, "arguments 必须是对象")

        handler = _TOOL_HANDLERS.get(str(name))
        if handler is None:
            return _error(rid, -32602, f"Unknown tool: {name}")

        try:
            data = handler(dict(arguments))
            text = _json_dumps(data)
            result: Dict[str, Any] = {
                "content": [{"type": "text", "text": text}],
                "structuredContent": data,
                "isError": False,
            }
            # 兼容验收：业务关键字段平铺到 result 顶层。
            if isinstance(data, Mapping):
                for k, v in data.items():
                    result.setdefault(k, v)
            return _ok(rid, result)
        except ToolError as exc:
            return _ok(
                rid,
                {
                    "content": [{"type": "text", "text": exc.message}],
                    "isError": True,
                    "errorCode": exc.code,
                    "error": exc.message,
                },
            )
        except Exception as exc:  # noqa: BLE001
            return _ok(
                rid,
                {
                    "content": [
                        {"type": "text", "text": f"{type(exc).__name__}: {exc}"}
                    ],
                    "isError": True,
                    "error": str(exc),
                },
            )

class McpHTTPServer(ThreadingHTTPServer):
    def __init__(self, addr, handler, mcp_server: McpServer):
        super().__init__(addr, handler)
        self.mcp_server = mcp_server


class _McpHTTPHandler(BaseHTTPRequestHandler):
    server_version = "TVPilotMCP/0.1"
    protocol_version = "HTTP/1.1"

    def log_message(self, fmt, *args):
        sys.stderr.write("[tvpilot-mcp] " + fmt % args + "\n")

    def _read_body(self) -> bytes:
        length = int(self.headers.get("Content-Length") or 0)
        return self.rfile.read(length) if length else b""

    def _send_json(self, obj: Any, status: int = 200) -> None:
        body = _json_dumps(obj).encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "application/json; charset=utf-8")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def _send_sse(self, obj: Any, status: int = 200) -> None:
        data = _json_dumps(obj)
        body = f"event: message\ndata: {data}\n\n".encode("utf-8")
        self.send_response(status)
        self.send_header("Content-Type", "text/event-stream; charset=utf-8")
        self.send_header("Cache-Control", "no-cache")
        self.send_header("Connection", "keep-alive")
        self.send_header("Content-Length", str(len(body)))
        self.end_headers()
        self.wfile.write(body)

    def do_GET(self):
        parsed = urlparse(self.path)

        if parsed.path.startswith(_SHOT_URL_PREFIX):
            fn = os.path.basename(parsed.path[len(_SHOT_URL_PREFIX):])
            path = os.path.join(_SHOT_DIR, fn)
            if not os.path.exists(path):
                self.send_error(404)
                return
            with open(path, "rb") as f:
                body = f.read()
            ctype = "image/png" if fn.lower().endswith(".png") else "image/jpeg"
            self.send_response(200)
            self.send_header("Content-Type", ctype)
            self.send_header("Content-Length", str(len(body)))
            self.end_headers()
            self.wfile.write(body)
            return

        if parsed.path == "/mcp":
            self._send_sse({"jsonrpc": "2.0", "method": "notifications/initialized", "params": {}})
            return

        self.send_error(404)

    def do_POST(self):
        parsed = urlparse(self.path)
        if parsed.path != "/mcp":
            self.send_error(404)
            return

        try:
            raw = self._read_body()
            req = json.loads(raw.decode("utf-8"))
        except Exception:
            self._send_json(_error(None, -32700, "Parse error"), 400)
            return

        server: McpServer = self.server.mcp_server  # type: ignore[attr-defined]
        resp = server.handle_jsonrpc(req)
        if resp is None:
            self.send_response(202)
            self.send_header("Content-Length", "0")
            self.end_headers()
            return

        accept = self.headers.get("Accept", "")
        if "text/event-stream" in accept:
            self._send_sse(resp)
        else:
            self._send_json(resp)


def serve_http(host: str = DEFAULT_HOST, port: int = DEFAULT_PORT) -> None:
    global _CURRENT_PORT
    mcp = McpServer(host, port)
    httpd = McpHTTPServer((host, port), _McpHTTPHandler, mcp)
    _CURRENT_PORT = int(httpd.server_address[1])
    sys.stderr.write(
        f"[tvpilot-mcp] HTTP/SSE listening on http://{host}:{_CURRENT_PORT}/mcp\n"
    )
    httpd.serve_forever()

def serve_stdio() -> None:
    server = McpServer()
    for line in sys.stdin:
        line = line.strip()
        if not line:
            continue
        try:
            req = json.loads(line)
        except Exception:
            sys.stdout.write(_json_dumps(_error(None, -32700, "Parse error")) + "\n")
            sys.stdout.flush()
            continue

        resp = server.handle_jsonrpc(req)
        if resp is not None:
            sys.stdout.write(_json_dumps(resp) + "\n")
            sys.stdout.flush()


def main(argv: Optional[List[str]] = None) -> None:
    parser = argparse.ArgumentParser(description="TVPilot MCP Server")
    parser.add_argument("--stdio", action="store_true", help="使用 stdio 传输")
    parser.add_argument("--host", default=DEFAULT_HOST)
    parser.add_argument("--port", type=int, default=DEFAULT_PORT)
    args = parser.parse_args(argv)

    if args.stdio:
        serve_stdio()
    else:
        serve_http(args.host, args.port)


if __name__ == "__main__":
    main()

5. §7 关键设计决策与未明确处
5.1 数据结构与索引选择
工具表 TOOLS 是静态列表，最多 10 个，直接线性查找，避免维护额外索引。

工具处理表 _TOOL_HANDLERS 是 name -> callable 字典，tools/call O(1) 查找。

截图文件放 tempfile.mkdtemp()，HTTP 模式通过 /screenshots/<file> 暴露；stdio 模式回退 file://。

BlueprintRunner 全局单例 + 锁，避免每次调用重复注册内置蓝图。

5.2 失败/降级语义
现有工具三元组 (data, err_code, err_msg)：有错误码或错误消息 => MCP isError: true，fail-closed。

二元组不按 (data, err) 解释，避免把 zap_via_adb_digits 这类 (ok, msg) 误判为失败。

工具内部异常 => MCP 工具错误，不向 JSON-RPC 层抛异常。

截图兼容调用：依次尝试 height=240、scale=240、quality="240p"、无参；若全部 TypeError，则报截图失败。

HTTP 通知 notifications/initialized 返回 202 空响应；stdio 通知不输出。

5.3 与 §1 现有实现的兼容边界
不修改任何现有文件。

tvpilot.tools、tvpilot.state、tvpilot.screenshot、tvpilot.blueprint 全部函数内延迟导入。

只调用需求点名的函数：tool_zap、tool_keyevent、tool_launch_app、tool_input_text、tool_tap、tool_combo_go_home、get_foreground_pkg、tool_system_state、tool_screenshot、BlueprintRunner。

tv_foreground 使用 get_foreground_pkg 构造 {"package": ...}，符合验收“无 ok/cost_ms 包装”。

5.4 需求说不清 / 需真仓确认的点
tvpilot/screenshot.py 未内联，tool_screenshot 的签名、返回结构未知。
本实现做了兼容探测：支持 URL、path、base64、dict 包装、三元组。若真仓返回结构特殊，需要按真仓调整 _screenshot_to_url_result。

tvpilot/state.py 的 tool_system_state 返回结构未内联。当前按三元组或直接数据透传。

MCP 2025-03-26 规范里 capabilities.tools 通常是对象，例如 {"listChanged": false}；但验收明确写 capabilities.tools=true，本实现按验收返回 true。若客户端严格按规范校验，可改为 {} 或 {"listChanged": false}。

HTTP/SSE 的 GET 流语义未明确。本实现 POST /mcp 在 Accept: text/event-stream 时返回 SSE；GET /mcp 返回一条保活 SSE 后关闭。

stdio 模式下截图 URL 使用 file://。若 agent 只能访问 HTTP URL，需要同时启动 HTTP 模式或额外静态服务。
