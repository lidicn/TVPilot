"""
tvpilottools.py - TVPilot 底层 ADB 控制工具
此文件为现有代码的接口定义，MCP Server 将调用这些函数。
"""
import subprocess
import json
import logging
import os
from typing import Optional

logger = logging.getLogger("tvpilot.tools")


class TVPilotTools:
    """Android TV ADB 控制工具集"""

    def __init__(self, device_ip: str = None, device_port: int = 5555):
        self.device_ip = device_ip or os.environ.get("TV_DEVICE_IP", "192.168.1.100")
        self.device_port = device_port
        self.adb_cmd_base = ["adb", "-s", f"{self.device_ip}:{self.device_port}"]

    def _run_adb(self, *args, timeout: int = 10) -> str:
        """执行 ADB 命令并返回 stdout"""
        cmd = self.adb_cmd_base + list(args)
        try:
            result = subprocess.run(
                cmd, capture_output=True, text=True, timeout=timeout
            )
            if result.returncode != 0:
                logger.error("ADB error: %s", result.stderr.strip())
                raise RuntimeError(f"ADB command failed: {result.stderr.strip()}")
            return result.stdout.strip()
        except subprocess.TimeoutExpired:
            raise RuntimeError(f"ADB command timed out after {timeout}s")

    def _run_shell(self, shell_cmd: str, timeout: int = 10) -> str:
        """执行 ADB shell 命令"""
        return self._run_adb("shell", shell_cmd, timeout=timeout)

    # ── 屏幕信息 ──
    def get_current_screen(self) -> dict:
        """获取当前前台应用信息"""
        output = self._run_shell(
            "dumpsys activity activities | grep -E 'mResumedActivity|mFocusedApp'"
        )
        # 解析 Activity 信息
        info = {"raw": output, "foreground_app": None, "activity": None}
        for line in output.splitlines():
            line = line.strip()
            if "mResumedActivity" in line or "mFocusedApp" in line:
                parts = line.split()
                for i, p in enumerate(parts):
                    if "/" in p and "." in p:
                        pkg_activity = p.rstrip("}")
                        if "/" in pkg_activity:
                            pkg, act = pkg_activity.split("/", 1)
                            info["foreground_app"] = pkg
                            info["activity"] = act
                        break
        return info

    # ── 应用启动 ──
    def launch_app(self, package_name: str) -> dict:
        """启动指定应用"""
        # 使用 monkey 启动（兼容性最好）
        output = self._run_shell(
            f"monkey -p {package_name} -c android.intent.category.LEANBACK_LAUNCHER 1"
        )
        if "No activities found" in output or "Error" in output:
            # fallback 到 am start
            output = self._run_shell(
                f"monkey -p {package_name} -c android.intent.category.LAUNCHER 1"
            )
        return {"package": package_name, "result": output}

    # ── 按键 ──
    def send_keyevent(self, keycode: str) -> dict:
        """发送按键事件"""
        output = self._run_shell(f"input keyevent {keycode}")
        return {"keycode": keycode, "result": output or "OK"}

    # ── 文本输入 ──
    def input_text(self, text: str) -> dict:
        """输入文本（ADB input text 空格用 %s 替代）"""
        escaped = text.replace(" ", "%s").replace("&", r"\&")
        output = self._run_shell(f"input text '{escaped}'")
        return {"text": text, "result": output or "OK"}

    # ── 音量 ──
    def set_volume(self, level: int, stream: str = "music") -> dict:
        """设置音量 (0-100)"""
        stream_map = {
            "music": 3, "system": 1, "ring": 2,
            "alarm": 4, "notification": 5,
        }
        stream_id = stream_map.get(stream, 3)
        # 先获取最大音量
        max_vol = int(self._run_shell(
            f"dumpsys audio | grep 'Stream {stream_id}' -A5 | grep 'Max' | head -1 | awk '{{print $2}}'"
        ) or "15")
        target = max(0, min(100, level))
        actual = int(target / 100 * max_vol)
        self._run_shell(f"media volume --set {actual} --stream {stream_id}")
        return {"level": target, "stream": stream, "actual_device_level": actual}

    def get_volume(self, stream: str = "music") -> dict:
        """获取当前音量"""
        stream_map = {
            "music": 3, "system": 1, "ring": 2,
            "alarm": 4, "notification": 5,
        }
        stream_id = stream_map.get(stream, 3)
        output = self._run_shell(
            f"dumpsys audio | grep 'Stream {stream_id}'"
        )
        return {"stream": stream, "raw": output}

    # ── 电源 ──
    def power_control(self, action: str) -> dict:
        """电源控制: on / off / toggle"""
        # 先检查屏幕状态
        screen_state = self._run_shell("dumpsys power | grep 'Display Power'")
        is_on = "state=ON" in screen_state

        if action == "on" and not is_on:
            self.send_keyevent("KEYCODE_WAKEUP")
        elif action == "off" and is_on:
            self.send_keyevent("KEYCODE_SLEEP")
        elif action == "toggle":
            self.send_keyevent("KEYCODE_POWER")

        return {"action": action, "was_on": is_on}

    # ── 已安装应用 ──
    def list_installed_apps(self) -> dict:
        """列出已安装的 Leanback/Launcher 应用"""
        output = self._run_shell("pm list packages -3")
        packages = [
            line.replace("package:", "").strip()
            for line in output.splitlines()
            if line.startswith("package:")
        ]
        return {"count": len(packages), "packages": sorted(packages)}

    # ── 截屏 ──
    def take_screenshot(self, save_dir: str = "/tmp/tvpilot_screenshots") -> dict:
        """截屏并保存到指定目录，返回文件路径"""
        import time
        os.makedirs(save_dir, exist_ok=True)
        ts = time.strftime("%Y%m%d_%H%M%S")
        remote_path = "/sdcard/tvpilot_screenshot.png"
        local_path = os.path.join(save_dir, f"screenshot_{ts}.png")

        self._run_shell(f"screencap -p {remote_path}")
        self._run_adb("pull", remote_path, local_path)
        self._run_shell(f"rm {remote_path}")

        return {"path": local_path, "timestamp": ts}

class TVPilotMCPServer:
    """
    TVPilot MCP Server —— 纯 stdio JSON-RPC 2.0 实现
    
    处理 MCP 协议消息:
      - initialize / initialized
      - tools/list
      - tools/call
      - ping
    
    所有日志输出到 stderr，JSON-RPC 通信走 stdin/stdout。
    """

    def __init__(self, tv_tools: TVPilotTools):
        self.tv = tv_tools
        self._initialized = False
        self._request_id_counter = 0

        # 工具名 → 处理函数 映射
        self._tool_handlers = {
            "get_screen_info": self._handle_get_screen_info,
            "launch_app": self._handle_launch_app,
            "send_key": self._handle_send_key,
            "input_text": self._handle_input_text,
            "set_volume": self._handle_set_volume,
            "get_volume": self._handle_get_volume,
            "power_control": self._handle_power_control,
            "list_installed_apps": self._handle_list_installed_apps,
            "screenshot": self._handle_screenshot,
        }

    # ────────────────────────────────────────
    # JSON-RPC 消息收发
    # ────────────────────────────────────────

    def run_stdio(self):
        """主循环：从 stdin 读取 JSON-RPC 消息，处理后写入 stdout"""
        logger.info("TVPilot MCP Server starting (stdio mode)...")
        
        # 确保 stdout 是 line-buffered
        stdout = TextIOWrapper(sys.stdout.buffer, encoding="utf-8", line_buffering=True)
        stdin = TextIOWrapper(sys.stdin.buffer, encoding="utf-8")

        for line in stdin:
            line = line.strip()
            if not line:
                continue
            try:
                message = json.loads(line)
            except json.JSONDecodeError:
                logger.warning("Invalid JSON received: %s", line[:200])
                continue

            logger.debug("Received: %s", json.dumps(message, ensure_ascii=False)[:500])

            # 区分 request / notification / response
            if "method" in message:
                if "id" in message:
                    # 这是一个 request，需要返回 response
                    response = self._handle_request(message)
                    if response is not None:
                        self._send_message(stdout, response)
                else:
                    # notification (无 id)，不需要响应
                    self._handle_notification(message)

    def _send_message(self, stdout, message: dict):
        """发送 JSON-RPC 消息到 stdout"""
        line = json.dumps(message, ensure_ascii=False)
        stdout.write(line + "\n")
        stdout.flush()
        logger.debug("Sent: %s", line[:500])

    # ────────────────────────────────────────
    # MCP 协议消息处理
    # ────────────────────────────────────────

    def _handle_request(self, msg: dict) -> Optional[dict]:
        """处理 JSON-RPC request，返回 response 或 None"""
        method = msg.get("method", "")
        params = msg.get("params", {})
        req_id = msg.get("id")

        handler_map = {
            "initialize": self._method_initialize,
            "tools/list": self._method_tools_list,
            "tools/call": self._method_tools_call,
            "ping": self._method_ping,
        }

        handler = handler_map.get(method)
        if handler:
            try:
                result = handler(params)
                return self._make_response(req_id, result)
            except MCPError as e:
                return self._make_error(req_id, e.code, e.message)
            except Exception as e:
                logger.exception("Unhandled error in method '%s'", method)
                return self._make_error(req_id, -32603, f"Internal error: {str(e)}")
        else:
            return self._make_error(req_id, -32601, f"Method not found: {method}")

    def _handle_notification(self, msg: dict):
        """处理 notification（无 id 的消息）"""
        method = msg.get("method", "")
        if method == "notifications/initialized":
            self._initialized = True
            logger.info("Client sent initialized notification")
        elif method == "notifications/cancelled":
            logger.info("Request cancelled: %s", msg.get("params", {}))
        else:
            logger.debug("Unhandled notification: %s", method)

    # ── MCP Methods ──

    def _method_initialize(self, params: dict) -> dict:
        """处理 initialize 请求"""
        logger.info(
            "Initialize request from client: %s",
            params.get("clientInfo", {}).get("name", "unknown"),
        )
        return {
            "protocolVersion": PROTOCOL_VERSION,
            "capabilities": {
                "tools": {"listChanged": False},
            },
            "serverInfo": {
                "name": SERVER_NAME,
                "version": SERVER_VERSION,
            },
        }

    def _method_tools_list(self, params: dict) -> dict:
        """返回可用工具列表"""
        return {"tools": TOOLS}

    def _method_tools_call(self, params: dict) -> dict:
        """执行工具调用"""
        tool_name = params.get("name", "")
        arguments = params.get("arguments", {})

        handler = self._tool_handlers.get(tool_name)
        if not handler:
            raise MCPError(
                -32602,
                f"Unknown tool: '{tool_name}'. "
                f"Available: {', '.join(self._tool_handlers.keys())}",
            )

        logger.info("Calling tool '%s' with args: %s", tool_name, list(arguments.keys()))

        try:
            result_data = handler(arguments)
        except MCPError:
            raise
        except Exception as e:
            logger.exception("Tool '%s' execution failed", tool_name)
            raise MCPError(-32603, f"Tool '{tool_name}' failed: {str(e)}")

        # 构造 MCP ToolResult
        return {
            "content": [
                {
                    "type": "text",
                    "text": json.dumps(result_data, ensure_ascii=False, indent=2),
                }
            ],
            "isError": False,
        }

    def _method_ping(self, params: dict) -> dict:
        """响应 ping"""
        return {}

    # ────────────────────────────────────────
    # 工具处理函数
    # ────────────────────────────────────────

    def _handle_get_screen_info(self, args: dict) -> dict:
        return self.tv.get_current_screen()

    def _handle_launch_app(self, args: dict) -> dict:
        pkg = args.get("package_name")
        if not pkg:
            raise MCPError(-32602, "Missing required parameter: package_name")
        return self.tv.launch_app(pkg)

    def _handle_send_key(self, args: dict) -> dict:
        keycode = args.get("keycode")
        if not keycode:
            raise MCPError(-32602, "Missing required parameter: keycode")
        return self.tv.send_keyevent(keycode)

    def _handle_input_text(self, args: dict) -> dict:
        text = args.get("text")
        if text is None:
            raise MCPError(-32602, "Missing required parameter: text")
        if not text:
            raise MCPError(-32602, "Parameter 'text' must not be empty")
        return self.tv.input_text(text)

    def _handle_set_volume(self, args: dict) -> dict:
        level = args.get("level")
        if level is None:
            raise MCPError(-32602, "Missing required parameter: level")
        if not isinstance(level, (int, float)) or level < 0 or level > 100:
            raise MCPError(-32602, "Parameter 'level' must be an integer between 0 and 100")
        stream = args.get("stream", "music")
        return self.tv.set_volume(int(level), stream)

    def _handle_get_volume(self, args: dict) -> dict:
        stream = args.get("stream", "music")
        return self.tv.get_volume(stream)

    def _handle_power_control(self, args: dict) -> dict:
        action = args.get("action")
        if action not in ("on", "off", "toggle"):
            raise MCPError(-32602, "Parameter 'action' must be 'on', 'off', or 'toggle'")
        return self.tv.power_control(action)

    def _handle_list_installed_apps(self, args: dict) -> dict:
        return self.tv.list_installed_apps()

    def _handle_screenshot(self, args: dict) -> dict:
        result = self.tv.take_screenshot()
        # 确保不返回 base64
        assert "base64" not in json.dumps(result).lower(), "Screenshot must not return base64"
        return result

    # ────────────────────────────────────────
    # 工具方法
    # ────────────────────────────────────────

    @staticmethod
    def _make_response(req_id, result: dict) -> dict:
        return {
            "jsonrpc": "2.0",
            "id": req_id,
            "result": result,
        }

    @staticmethod
    def _make_error(req_id, code: int, message: str) -> dict:
        return {
            "jsonrpc": "2.0",
            "id": req_id,
            "error": {"code": code, "message": message},
        }


class MCPError(Exception):
    """MCP 协议错误"""
    def __init__(self, code: int, message: str):
        self.code = code
        self.message = message
        super().__init__(message)

    # 忽略 SIGPIPE，防止客户端断开时崩溃
    if hasattr(signal, "SIGPIPE"):
        signal.signal(signal.SIGPIPE, signal.SIG_IGN)

    if not args.device_ip:
        logger.error(
            "未指定设备 IP！请通过 --device-ip 参数或 TV_DEVICE_IP 环境变量设置。"
        )
        sys.exit(1)

    # 初始化工具层
    tv_tools = TVPilotTools(device_ip=args.device_ip, device_port=args.device_port)

    # 启动 MCP Server
    server = TVPilotMCPServer(tv_tools)
    
    try:
        server.run_stdio()
    except KeyboardInterrupt:
        logger.info("Server shutting down (KeyboardInterrupt)")
    except BrokenPipeError:
        logger.info("Client disconnected (BrokenPipe)")
    except Exception:
        logger.exception("Server crashed")
        sys.exit(1)


if __name__ == "__main__":
    main()
