3.3 工具 handler（关键实现）
----------------------------

    # ---------- 返回值归一化 ----------
    # 现有工具返回三元组 (data, err_code, err_msg)；
    # MCP 要求只返回 data（无 ok/cost_ms 包装），出错时返回 isError=True。

    def _normalize(result: Tuple) -> Tuple[Any, Optional[str]]:
        """把三元组归一化为 (data, error_message)。"""
        data, err_code, err_msg = normalize_tool_result(result)
        if err_code is not None or err_msg is not None:
            return None, err_msg or err_code
        return data, None


    # ---------- 各工具 handler ----------

    def _tool_foreground(args: Dict[str, Any]) -> Dict[str, Any]:
        """返回 {"package": "..."}"""
        data, err = _normalize(tool_foreground())
        if err:
            return _error_result(err)
        return data  # {"package": "..."}


    def _tool_screenshot(args: Dict[str, Any]) -> Dict[str, Any]:
        """返回 {"url": "...", "width": 426}（不返回 base64）。"""
        quality = args.get('quality', '240p')
        # 240p 对应 426x240（16:9 比例，标准 240p）
        if quality == '240p':
            width = 426
            height = 240
        else:
            width = 854
            height = 480

        data, err = _normalize(tool_screenshot(width=width, height=height))
        if err:
            return _error_result(err)
        # 确保只返回 url + width，不返回 base64
        result = {}
        if isinstance(data, dict):
            result['url'] = data.get('url', '')
            result['width'] = data.get('width', width)
        return result


    def _tool_zap(args: Dict[str, Any]) -> Dict[str, Any]:
        """返回换台结果关键字段。"""
        channel = args.get('channel', '')
        data, err = _normalize(tool_zap(channel))
        if err:
            return _error_result(err)
        return data


    def _tool_keyevent(args: Dict[str, Any]) -> Dict[str, Any]:
        """返回按键结果。"""
        key = args.get('key', '')
        data, err = _normalize(tool_keyevent(key))
        if err:
            return _error_result(err)
        return data


    def _tool_tap(args: Dict[str, Any]) -> Dict[str, Any]:
        """返回点击结果。"""
        x = args.get('x', 0)
        y = args.get('y', 0)
        data, err = _normalize(tool_tap(x, y))
        if err:
            return _error_result(err)
        return data


    def _tool_input_text(args: Dict[str, Any]) -> Dict[str, Any]:
        """返回输入结果。"""
        text = args.get('text', '')
        data, err = _normalize(tool_input_text(text))
        if err:
            return _error_result(err)
        return data


    def _tool_launch_app(args: Dict[str, Any]) -> Dict[str, Any]:
        """返回启动 App 结果。"""
        package = args.get('package', '')
        activity = args.get('activity')
        data, err = _normalize(tool_launch_app(package, activity))
        if err:
            return _error_result(err)
        return data


    def _tool_go_home(args: Dict[str, Any]) -> Dict[str, Any]:
        """返回回桌面结果。"""
        data, err = _normalize(tool_combo_go_home())
        if err:
            return _error_result(err)
        return data


    def _tool_system_state(args: Dict[str, Any]) -> Dict[str, Any]:
        """返回系统状态。"""
        data, err = _normalize(tool_system_state())
        if err:
            return _error_result(err)
        return data


    def _tool_blueprint_run(args: Dict[str, Any]) -> Dict[str, Any]:
        """调用 BlueprintRunner 执行蓝图，返回汇总结果。"""
        name = args.get('name', '')
        params = args.get('params', {})
        runner = BlueprintRunner()
        runner.register_builtins()
        result = runner.run(name, params)
        # BlueprintResult 有 to_dict() 方法
        if hasattr(result, 'to_dict'):
            d = result.to_dict()
            # 精简返回：只保留关键字段
            return {
                'name': d.get('name'),
                'ok': d.get('ok'),
                'error': d.get('error'),
                'steps': [
                    {
                        'step': s.get('step'),
                        'ok': s.get('ok'),
                        'error': s.get('error'),
                    }
                    for s in d.get('steps', [])
                ],
            }
        return result


3.4 JSON-RPC 2.0 Dispatcher
---------------------------

    MCP_PROTOCOL_VERSION = '2025-03-26'


    class JsonRpcError:
        """JSON-RPC 2.0 错误码。"""
        PARSE_ERROR = -32700
        INVALID_REQUEST = -32600
        METHOD_NOT_FOUND = -32601
        INVALID_PARAMS = -32602
        INTERNAL_ERROR = -32603


    def _error_result(msg: str) -> Dict[str, Any]:
        """构造工具调用错误响应（isError=True）。"""
        return {'isError': True, 'error': msg}


    def _jsonrpc_error(code: int, msg: str) -> Dict[str, Any]:
        """构造 JSON-RPC 错误响应。"""
        return {
            'jsonrpc': '2.0',
            'error': {'code': code, 'message': msg},
        }


    def handle_request(body: bytes, tools: List[MCPTool]) -> Dict[str, Any]:
        """处理 JSON-RPC 2.0 请求，返回响应 dict。"""
        try:
            req = json.loads(body)
        except json.JSONDecodeError as e:
            return _jsonrpc_error(JsonRpcError.PARSE_ERROR, str(e))

        if not isinstance(req, dict):
            return _jsonrpc_error(JsonRpcError.INVALID_REQUEST, 'Request must be a JSON object')

        method = req.get('method')
        req_id = req.get('id')
        params = req.get('params', {})

        if method is None:
            return _jsonrpc_error(JsonRpcError.INVALID_REQUEST, 'Missing method')

        # ---- 方法分发 ----

        if method == 'initialize':
            result = {
                'protocolVersion': MCP_PROTOCOL_VERSION,
                'capabilities': {
                    'tools': True,
                },
                'serverInfo': {
                    'name': 'tvpilot-mcp',
                    'version': '0.1.0',
                },
            }
            return {'jsonrpc': '2.0', 'id': req_id, 'result': result}

        elif method == 'tools/list':
            tool_list = []
            for t in tools:
                tool_list.append({
                    'name': t.name,
                    'description': t.description,
                    'inputSchema': t.input_schema,
                })
            return {'jsonrpc': '2.0', 'id': req_id, 'result': {'tools': tool_list}}

        elif method == 'tools/call':
            tool_name = params.get('name', '')
            tool_args = params.get('arguments', {})

            tool_map = {t.name: t for t in tools}
            tool = tool_map.get(tool_name)
            if tool is None:
                return _jsonrpc_error(
                    JsonRpcError.INVALID_PARAMS,
                    f'Unknown tool: {tool_name}')

            try:
                data = tool.handler(tool_args)
            except Exception as exc:
                return _jsonrpc_error(
                    JsonRpcError.INTERNAL_ERROR,
                    f'{type(exc).__name__}: {exc}')

            # 检查是否为工具层错误
            if isinstance(data, dict) and data.get('isError'):
                return {
                    'jsonrpc': '2.0',
                    'id': req_id,
                    'result': {
                        'content': [{'type': 'text', 'text': json.dumps(
                            {'error': data.get('error', data.get('message', '')))},
                            ensure_ascii=False)}],
                        'isError': True,
                    },
                }

            # 正常返回
            return {
                'jsonrpc': '2.0',
                'id': req_id,
                'result': {
                    'content': [{'type': 'text', 'text': json.dumps(
                        data, ensure_ascii=False, default=str)}],
                },
            }

        else:
            return _jsonrpc_error(
                JsonRpcError.METHOD_NOT_FOUND,
                f'Method not found: {method}')


3.5 HTTP/SSE Transport（POST /mcp）
-----------------------------------

    class MCPHTTPHandler(BaseHTTPRequestHandler):
        """HTTP handler for MCP JSON-RPC requests."""
        tools: List[MCPTool] = []

        def do_POST(self) -> None:
            if urlparse(self.path).path != '/mcp':
                self.send_response(404)
                self.end_headers()
                return

            length = int(self.headers.get('Content-Length', 0))
            body = self.rfile.read(length)

            resp = handle_request(body, self.tools)
            resp_bytes = json.dumps(resp, ensure_ascii=False).encode('utf-8')

            self.send_response(200)
            self.send_header('Content-Type', 'application/json; charset=utf-8')
            self.send_header('Content-Length', str(len(resp_bytes)))
            self.end_headers()
            self.wfile.write(resp_bytes)

        def log_message(self, format: str, *args: Any) -> None:
            pass  # 静默日志，避免污染 stdio 输出

    def run_http_server(host: str = '0.0.0.0', port: int = 8765) -> None:
        """启动 HTTP/SSE MCP 服务器。"""
        tools = build_tool_registry()
        MCPHTTPHandler.tools = tools
        server = ThreadingHTTPServer((host, port), MCPHTTPHandler)
        print(f'MCP HTTP Server listening on http://{host}:{port}/mcp')
        server.serve_forever()


3.6 Stdio Transport
-------------------

    def run_stdio() -> None:
        """stdio 模式：从 stdin 读 JSON-RPC 请求，写 stdout 响应。

        每行一条 JSON-RPC 消息（MCP stdio 约定）。
        """
        tools = build_tool_registry()

        for line in sys.stdin:
            line = line.strip()
            if not line:
                continue

            try:
                req = json.loads(line)
            except json.JSONDecodeError:
                resp = _jsonrpc_error(JsonRpcError.PARSE_ERROR, 'Invalid JSON')
                sys.stdout.write(json.dumps(resp, ensure_ascii=False) + '\n')
                sys.stdout.flush()
                continue

            # 处理请求
            body = json.dumps(req).encode('utf-8')
            resp = handle_request(body, tools)

            # 写响应
            sys.stdout.write(json.dumps(resp, ensure_ascii=False) + '\n')
            sys.stdout.flush()


3.7 入口函数（__main__）

    def main() -> None:
        """CLI 入口：python -m tvpilot.mcp --stdio 或 python -m tvpilot.mcp --http。"""
        if '--stdio' in sys.argv:
            run_stdio()
        elif '--http' in sys.argv:
            port = 8765
            for i, arg in enumerate(sys.argv):
                if arg == '--port' and i + 1 < len(sys.argv):
                    port = int(sys.argv[i + 1])
            run_http_server(port=port)
        else:
            print('Usage: python -m tvpilot.mcp --stdio | --http [--port PORT]')
            sys.exit(1)


    if __name__ == '__main__':
        main()


4. 设计决策记录
===============

4.1 数据结构与索引选择
----------------------
- 工具注册表使用 List[MCPTool]（有序列表），分发时用 dict 映射加速查找。
- 原因：MCP 协议要求 tools/list 返回有序列表；内部分发用 dict 保证 O(1) 查找。
- 每个 MCPTool 有 4 个字段（name/description/input_schema/handler），用 __slots__ 减少内存。

4.2 失败/降级语义
-----------------
- 所有工具 handler 一律 fail-closed：出错返回 isError=True 而非抛异常。
- 原因：MCP 客户端需要能处理工具错误而不崩溃；抛异常会导致 JSON-RPC 层返回 INTERNAL_ERROR，
  客户端无法区分"电视离线"和"服务器 bug"。
- JSON-RPC 层只在以下情况返回 error：请求解析失败、方法不存在、参数非法。
  工具调用失败返回 isError=True 的 result（MCP 工具调用语义）。

4.3 与 §1 现有实现的兼容边界
-----------------------------
- 所有工具 handler 通过 import 调用现有函数，不复制不修改任何代码。
- 返回值归一化：现有工具返回 (data, err_code, err_msg) 三元组，MCP 层只取 data；
  err_code/err_msg 非 None 时转换为 isError=True。
- 截图工具：默认 240p（426x240），可选 480p（854x480），始终返回 URL 不返回 base64。
  原因：LLM token 节省（base64 一张 480p JPEG ≈ 50KB ≈ 12.5K tokens）。

4.4 需求中说不清的地方
----------------------
- screenshot.tool_screenshot 的确切参数签名未在需求稿中给出。本设计假设它接受
  width/height 关键字参数并返回 ({"url":..., "width":...}, None, None) 三元组。
  落地前需确认实际签名。
- BlueprintRunner 是否需要初始化时传入参数（如 tools 表、sleep 函数）。本设计假设
  无参构造 + register_builtins() 即可使用。
- state.tool_system_state 的确切返回字段未明确。本设计透传其返回值。
- HTTP 端口默认 8765，需求稿未指定端口号。

    # T8: stdio 模式响应 initialize
    def test_stdio_initialize():
        import subprocess
        proc = subprocess.Popen(
            [sys.executable, '-m', 'tvpilot.mcp', '--stdio'],
            stdin=subprocess.PIPE, stdout=subprocess.PIPE,
            stderr=subprocess.PIPE, text=True,
        )
        req = json.dumps({'jsonrpc': '2.0', 'id': 1, 'method': 'initialize', 'params': {}})
        proc.stdin.write(req + '\n')
        proc.stdin.flush()
        # 发送 EOF 关闭 stdin
        proc.stdin.close()
        stdout, stderr = proc.communicate(timeout=10)
        resp = json.loads(stdout.strip().split('\n')[0])
        assert resp['result']['protocolVersion'] == '2025-03-26'

    # T12: 零新依赖——import 列表只有标准库 + tvpilot
    def test_no_new_dependencies():
        import ast
        import importlib
        stdlib = set(sys.stdlib_module_names)
        allowed = stdlib | {'tvpilot', 'tvpilot.tools', 'tvpilot.state',
                            'tvpilot.screenshot', 'tvpilot.blueprint'}
        # 静态分析 mcp.py 的 import 列表
        with open(__file__) as f:  # 假设测试文件可访问 mcp.py
            tree = ast.parse(open('tvpilot/mcp.py').read())
        for node in ast.walk(tree):
            if isinstance(node, ast.Import):
                for alias in node.names:
                    top = alias.name.split('.')[0]
                    assert top in allowed, f'New dependency: {top}'
            elif isinstance(node, ast.ImportFrom):
                if node.module:
                    top = node.module.split('.')[0]
                    assert top in allowed, f'New dependency: {top}'
